Código Python con openpyxl

Si prefieres ejecutar este procedimiento de forma automatizada en tu servidor o consola local, utiliza el siguiente script:

import openpyxl
from copy import copy

def fusionar_cantos(file_v1="CantosOrdenados.xlsx", file_v2="CantosOrdenadosV2.xlsx", file_out="CantosOrdenadosV3.xlsx"):
    """
    Fusiona CantosOrdenados.xlsx y CantosOrdenadosV2.xlsx manteniendo:
    - Estructura exacta, orden y nombres de columnas de V1.
    - Estilos visuales (fuentes, rellenos, bordes, alineación, formato numérico).
    - Mantiene todos los registros duplicados.
    """
    print("Cargando archivos Excel...")
    wb1 = openpyxl.load_workbook(file_v1)
    wb2 = openpyxl.load_workbook(file_v2)
    
    ws1 = wb1.active
    ws2 = wb2.active

    # 1. Obtener mapeo de encabezados de V1
    headers_v1 = [cell.value for cell in ws1[1]]
    
    # 2. Obtener encabezados de V2
    headers_v2 = [cell.value for cell in ws2[1]]
    
    # Mapeo de nombre de columna a índice en V2
    map_v2 = {}
    for idx, h in enumerate(headers_v2):
        if h is not None:
            map_v2[str(h).strip().lower()] = idx + 1

    # Copiar estilos de celda desde la última fila con datos de V1 (o fila 2 si existe)
    sample_row_idx = ws1.max_row if ws1.max_row >= 2 else 2

    # 3. Leer filas de V2 e insertarlas en V1
    filas_agregadas = 0
    for row_idx in range(2, ws2.max_row + 1):
        # Verificar si la fila de V2 no está completamente vacía
        row_values = [ws2.cell(row=row_idx, column=c).value for c in range(1, len(headers_v2) + 1)]
        if not any(v is not None for v in row_values):
            continue

        target_row = ws1.max_row + 1
        
        for col_v1_idx, header_name in enumerate(headers_v1, start=1):
            target_cell = ws1.cell(row=target_row, column=col_v1_idx)
            
            # Buscar valor correspondiente en V2 por nombre de columna
            norm_header = str(header_name).strip().lower() if header_name else ""
            if norm_header in map_v2:
                col_v2_idx = map_v2[norm_header]
                val = ws2.cell(row=row_idx, column=col_v2_idx).value
                target_cell.value = val
            else:
                target_cell.value = None

            # Copiar estilos de la fila de muestra en V1
            sample_cell = ws1.cell(row=sample_row_idx, column=col_v1_idx)
            if sample_cell.has_style:
                target_cell.font = copy(sample_cell.font)
                target_cell.fill = copy(sample_cell.fill)
                target_cell.border = copy(sample_cell.border)
                target_cell.alignment = copy(sample_cell.alignment)
                target_cell.number_format = sample_cell.number_format

        filas_agregadas += 1

    # 4. Guardar resultado final
    wb1.save(file_out)
    print(f"¡Éxito! Se integraron {filas_agregadas} registros de V2 en {file_out}.")

if __name__ == "__main__":
    fusionar_cantos()
